53def _run_generation( 

54 task_id: str, 

55 params: VideoParams, 

56 capture_logs: bool,

60 voxcpm_prompt_audio: bytes | None = None,

62) -> dict: 

63 """ 

64 在后台线程中执行现有视频流水线。 

65 

66 Loguru 的 sink 是进程级资源，因此必须按当前工作线程过滤。否则同时运行的 

67 API 任务或其它页面日志会混入当前任务。任务为下载、片段编码和心跳启动的 

68 辅助线程通过 bind_log_scope 归属到工作线程，它们的日志同样要收集，否则 

69 这些耗时阶段在 WebUI 里没有任何输出。页面只读取普通列表快照，不会从后台 

70 线程访问 Streamlit session_state，从根源上避免刷新时的 delta 路径错乱。 

71 """ 

72 log_handler_id = None 

73 worker_thread_id = threading.get_ident() 

74 try: 

75 if capture_logs: 

76 log_handler_id = logger.add( 

77 lambda message: _append_task_log(task_id, str(message)), 

78 level="DEBUG", 

79 format=format_log_record, 

80 colorize=False, 

81 filter=lambda record: ( 

82 log_scope_thread_id(record["thread"].id) == worker_thread_id 

83 ), 

84 ) 

85 

86 # 完整任务仍使用原来的配置锁，防止另一个 WebUI 会话在生成中途修改 

87 # Provider、密钥等进程级配置，造成同一条视频前后使用不同设置。 

88 with config.runtime_config_lock(): 

89 return tm.start( 

90 task_id=task_id, 

91 params=params, 

92 voice_preview=voice_preview, 

93 loomloom_video_request=loomloom_video_request, 

94 voxcpm_reference_audio=voxcpm_reference_audio, 

95 voxcpm_prompt_audio=voxcpm_prompt_audio, 

96 voxcpm_prompt_text=voxcpm_prompt_text, 

97 ) 

98 except Exception as exc: 

99 # tm.start 已负责把流水线异常转换成失败状态；这里额外保护日志 sink、 

100 # 配置锁等 WebUI 包装层。任何后台线程异常都必须留下终态，不能让任务 

101 # 管理器在工作线程退出后仍永久显示“生成中”。 

102 error = f"{type(exc).__name__}: {exc}" 

103 failure = { 

104 "task_id": task_id, 

105 "state": const.TASK_STATE_FAILED, 

106 "progress": 0, 

107 "failed_stage": "webui_worker", 

108 "error": error, 

109 }

112 state=failure["state"],

114 failed_stage=failure["failed_stage"], 

115 error=failure["error"], 

116 )

119 f"task_id={task_id}, error={exc}" 

120 )

127 logger.debug(
