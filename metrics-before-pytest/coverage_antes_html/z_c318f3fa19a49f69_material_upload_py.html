15# Local materials are usually short clips. This matches Streamlit's default upload 

16# limit while still placing an explicit server-side bound on direct API clients. 

17MAX_VIDEO_MATERIAL_UPLOAD_BYTES = 200 * 1024 * 1024 

18MAX_IMAGE_MATERIAL_UPLOAD_BYTES = 20 * 1024 * 1024 

19MATERIAL_VALIDATION_TIMEOUT_SECONDS = 120

21# Keep these aligned with the local material formats the CLI accepts (cli.py 

22# derives its list from const.FILE_TYPE_VIDEOS / const.FILE_TYPE_IMAGES) and the 

23# render pipeline classifies (video.py reads const.FILE_TYPE_IMAGES). Accepting a 

24# narrower set here rejects uploads the rest of the pipeline already supports. 

25SUPPORTED_VIDEO_EXTENSIONS = (".mp4", ".mov", ".avi", ".flv", ".mkv", ".webm") 

26SUPPORTED_IMAGE_EXTENSIONS = (".jpg", ".jpeg", ".png", ".bmp") 

27SUPPORTED_MATERIAL_EXTENSIONS = ( 

28 *SUPPORTED_VIDEO_EXTENSIONS, 

29 *SUPPORTED_IMAGE_EXTENSIONS, 

30) 

31 

32_COPY_CHUNK_BYTES = 1024 * 1024 

33_INTERNAL_UPLOAD_PREFIX = ".material-upload-" 

34_WINDOWS_INVALID_FILENAME_CHARS = frozenset('<>:"|?*') 

35_WINDOWS_RESERVED_FILENAMES = frozenset( 

36 {"CON", "PRN", "AUX", "NUL"} 

37 | {f"{prefix}{number}" for prefix in ("COM", "LPT") for number in range(1, 10)} 

38 # 与 bgm.sanitize_upload_filename 及 webui/Main.py 的下载文件名规则保持同一份 

39 # 官方清单：Win32 把 Latin-1 上标数字 ¹、²、³ 也当作设备编号。 

40 | {f"{prefix}{number}" for prefix in ("COM", "LPT") for number in ("¹", "²", "³")} 

41) 

42# 文件名会原样进入日志、API 响应和 WebUI 界面，因此除路径分隔符与 Win32 保留名外， 

43# 还要拒绝所有控制符和会改变显示形态的字符。此前只拦截 ord < 32（C0 控制符）， 

44# 漏掉了同一类问题的另一半： 

45# * C1 控制符 U+007F-U+009F：U+0085 会被部分日志查看器渲染成换行，一个文件名就能 

46# 伪造出额外的、看起来独立的日志行。 

47# * 双向文本控制符 U+200E/U+200F/U+202A-U+202E/U+2066-U+2069：U+202E 之后的字符会被 

48# 反向渲染，"photo\u202egnp.mp4" 在资源管理器和日志里看起来像另一个扩展名。 

49# * U+2028/U+2029（Unicode 行/段分隔符）同样能在一行日志里制造视觉换行。 

50# 这些字符都无法在文件名输入框里键入，正常上传不受影响；判定与 

51# bgm.sanitize_upload_filename、app/controllers/base.py 的 normalize_task_id 一致。 

52_UNSAFE_FILENAME_CHARACTERS = frozenset( 

53 chr(code) 

54 for code in ( 

55 *range(0x00, 0x20), 

56 *range(0x7F, 0xA0), 

57 0x200E, 

58 0x200F, 

59 0x2028, 

60 0x2029, 

61 *range(0x202A, 0x202F), 

62 *range(0x2066, 0x206A), 

63 ) 

64) 

65_IMAGE_FORMATS_BY_EXTENSION = { 

66 ".jpg": frozenset({"JPEG"}), 

67 ".jpeg": frozenset({"JPEG"}), 

68 ".png": frozenset({"PNG"}), 

69 ".bmp": frozenset({"BMP"}),

97def _material_kind(filename: str) -> Literal["video", "image"]: 

98 suffix = Path(filename).suffix.lower() 

99 if suffix in SUPPORTED_VIDEO_EXTENSIONS:

101 if suffix in SUPPORTED_IMAGE_EXTENSIONS:

112def sanitize_material_filename(filename: str) -> str: 

113 """Return a display-safe basename and reject malformed or unsupported names.""" 

114 safe_name = (filename or "").replace("\\", "/").split("/")[-1].strip() 

115 if ( 

116 not safe_name 

117 or safe_name in {".", ".."} 

118 or len(safe_name) > 255 

119 or any(character in _UNSAFE_FILENAME_CHARACTERS for character in safe_name) 

120 or any(character in _WINDOWS_INVALID_FILENAME_CHARS for character in safe_name) 

121 or safe_name.lower().startswith(_INTERNAL_UPLOAD_PREFIX) 

122 ): 

123 raise MaterialUploadError("invalid local material filename") 

124 

125 # Keep the same rule as bgm.sanitize_upload_filename: Windows resolves the 

126 # segment before the extension as a device name, so CON.mp4 and LPT1.webm 

127 # cannot be created as ordinary files there. Even though the stored name is 

128 # a UUID in both endpoints, rejecting these names up front keeps the two 

129 # upload APIs behaving identically on every platform. 

130 windows_basename = safe_name.split(".", 1)[0].rstrip(" .").upper() 

131 if windows_basename in _WINDOWS_RESERVED_FILENAMES: 

132 raise MaterialUploadError("invalid local material filename") 

133 _material_kind(safe_name) 

134 return safe_name

137def _validate_image(file_path: str, extension: str) -> None: 

138 try: 

139 # Pillow protects against excessively large pixel dimensions. Treat its 

140 # warning threshold as an upload error too, rather than only rejecting at 

141 # the higher DecompressionBombError threshold. 

142 with warnings.catch_warnings(): 

143 warnings.simplefilter("error", Image.DecompressionBombWarning) 

144 with Image.open(file_path) as image: 

145 image_format = str(image.format or "").upper() 

146 if image_format not in _IMAGE_FORMATS_BY_EXTENSION[extension]: 

147 raise MaterialUploadError( 

148 "uploaded image content does not match its file extension" 

149 ) 

150 image.verify() 

151 # verify() checks structure but may accept truncated JPEG pixel data. 

152 # Reopen after verify() (which invalidates the decoder) and decode 

153 # every pixel before publishing the upload to later render tasks. 

154 with Image.open(file_path) as image: 

155 image.load() 

156 except MaterialUploadError: 

157 raise 

158 except ( 

159 Image.DecompressionBombError, 

160 Image.DecompressionBombWarning, 

161 UnidentifiedImageError, 

162 OSError, 

163 SyntaxError, 

164 ValueError, 

165 ) as exc: 

166 raise MaterialUploadError( 

167 "uploaded file must contain a valid JPEG, PNG, or BMP image" 

168 ) from exc

171def _validate_video( 

172 file_path: str, timeout_seconds: int = MATERIAL_VALIDATION_TIMEOUT_SECONDS 

173) -> None: 

174 # FFmpeg treats a standalone image as a one-frame video stream. Reject images 

175 # explicitly so renaming photo.jpg to photo.mp4 cannot bypass the media class. 

176 try: 

177 with warnings.catch_warnings(): 

178 warnings.simplefilter("error", Image.DecompressionBombWarning) 

179 with Image.open(file_path) as image: 

180 image.verify() 

181 except (Image.DecompressionBombError, Image.DecompressionBombWarning) as exc: 

182 raise MaterialUploadError( 

183 "uploaded file must contain a video, not an image" 

184 ) from exc 

185 except (UnidentifiedImageError, OSError, SyntaxError, ValueError): 

186 pass 

187 else: 

188 raise MaterialUploadError("uploaded file must contain a video, not an image") 

189 

190 try: 

191 decoded = subprocess.run( 

192 [ 

193 utils.get_ffmpeg_binary(), 

194 "-nostdin", 

195 "-v", 

196 "error", 

197 "-xerror", 

198 # Uploaded files must be self-contained media, not playlists 

199 # that cause validation/rendering to read adjacent files or URLs. 

200 "-protocol_whitelist", 

201 "file,pipe", 

202 "-format_whitelist", 

203 "mov,avi,flv,matroska", 

204 "-i", 

205 file_path, 

206 "-map", 

207 # Uppercase V excludes attached pictures/album covers. A cover 

208 # in an audio container is not a renderable video material. 

209 "0:V:0", 

210 "-f", 

211 "null", 

212 "-", 

213 ], 

214 capture_output=True, 

215 timeout=timeout_seconds, 

216 check=False, 

217 ) 

218 except subprocess.TimeoutExpired as exc: 

219 raise MaterialServiceError("FFmpeg material validation timed out") from exc 

220 except OSError as exc: 

221 raise MaterialServiceError( 

222 "failed to run FFmpeg for material validation" 

223 ) from exc 

224 if decoded.returncode != 0: 

225 raise MaterialUploadError( 

226 "uploaded file must contain a completely decodable video stream" 

227 )

230def _stage_material_upload( 

231 filename: str, source: BinaryIO 

232) -> tuple[str, Literal["video", "image"], str, int]: 

233 safe_name = sanitize_material_filename(filename) 

234 material_kind = _material_kind(safe_name) 

235 maximum_bytes = ( 

236 MAX_VIDEO_MATERIAL_UPLOAD_BYTES 

237 if material_kind == "video" 

238 else MAX_IMAGE_MATERIAL_UPLOAD_BYTES 

239 ) 

240 maximum_megabytes = maximum_bytes // (1024 * 1024) 

241 

242 try: 

243 target_dir = uploaded_material_dir(create=True) 

244 except OSError as exc: 

245 raise MaterialServiceError("failed to prepare local material storage") from exc 

246 

247 temp_path = "" 

248 total_bytes = 0 

249 try: 

250 try: 

251 source.seek(0) 

252 except (AttributeError, OSError) as exc: 

253 raise MaterialUploadError("local material upload is not seekable") from exc 

254 

255 descriptor, temp_path = tempfile.mkstemp( 

256 prefix=_INTERNAL_UPLOAD_PREFIX, 

257 suffix=Path(safe_name).suffix.lower(), 

258 dir=target_dir, 

259 ) 

260 with os.fdopen(descriptor, "wb") as output: 

261 while True: 

262 chunk = source.read(_COPY_CHUNK_BYTES) 

263 if not chunk: 

264 break 

265 if not isinstance(chunk, (bytes, bytearray, memoryview)): 

266 raise MaterialUploadError("local material upload must be binary") 

267 total_bytes += len(chunk) 

268 if total_bytes > maximum_bytes: 

269 raise MaterialUploadError( 

270 f"{material_kind} material exceeds the " 

271 f"{maximum_megabytes} MB limit" 

272 ) 

273 output.write(chunk) 

274 output.flush() 

275 os.fsync(output.fileno()) 

276 

277 if total_bytes == 0: 

278 raise MaterialUploadError("local material file is empty") 

279 return safe_name, material_kind, temp_path, total_bytes 

280 except Exception as exc: 

281 _remove_staged_file(temp_path) 

282 if isinstance(exc, MaterialUploadError): 282 ↛ 284line 282 didn't jump to line 284 because the condition on line 282 was always true

283 raise 

284 if isinstance(exc, OSError): 

285 raise MaterialServiceError("failed to stage local material upload") from exc 

286 raise 

287 finally: 

288 try: 

289 source.seek(0) 

290 except (AttributeError, OSError): 

291 pass

294def save_material_upload(filename: str, source: BinaryIO) -> str: 

295 """Validate and atomically persist an uploaded local video or image.""" 

296 safe_name, material_kind, temp_path, total_bytes = _stage_material_upload( 

297 filename, source 

298 ) 

299 extension = Path(safe_name).suffix.lower() 

300 stored_name = f"{uuid4().hex}{extension}" 

301 target_path = os.path.join(os.path.dirname(temp_path), stored_name) 

302 

303 try: 

304 if material_kind == "video": 

305 _validate_video(temp_path) 

306 else: 

307 _validate_image(temp_path, extension) 

308 

309 try: 

310 os.replace(temp_path, target_path) 

311 except OSError as exc: 

312 raise MaterialServiceError( 

313 "failed to persist local material upload" 

314 ) from exc 

315 temp_path = "" 

316 logger.info( 

317 f"local material uploaded: original_name={safe_name}, " 

318 f"stored_name={stored_name}, kind={material_kind}, " 

319 f"size={total_bytes} bytes" 

320 ) 

321 return stored_name 

322 finally: 

323 _remove_staged_file(temp_path)
