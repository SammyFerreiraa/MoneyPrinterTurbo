88def _normalize_allowed_origin(raw_origin: str) -> str | None: 

89 """把配置项折叠成浏览器实际发送的 Origin 形式。 

90 

91 Origin 请求头由 RFC 6454 固定为 ``scheme://host[:port]``：没有路径、没有尾 

92 斜杠，scheme 与 host 已经规范化为小写，并且省略协议的默认端口。用户在浏览器 

93 地址栏复制得到的 ``https://frontend.example/``，以及显式写出默认端口的 

94 ``https://frontend.example:443``，都与浏览器实际发送的 

95 ``https://frontend.example`` 逐字符不相等，白名单因此静默失效：前端只看到 

96 CORS 报错，服务端只留下一行 blocked 日志，两边都指不到配置本身。 

97 

98 返回 ``None`` 表示该写法不可能匹配任何 Origin（缺少 scheme、scheme 不是 

99 http/https、主机名为空或含空白、IPv6 字面量畸形、端口非法或越界），调用方应 

100 丢弃并告警。解析失败必须收敛成 ``None``：本函数在模块导入期就会执行，让 

101 ``urlsplit()`` 的 ``ValueError`` 冒出去等于整条 API 无法启动。 

102 """ 

103 

104 if raw_origin == "*": 

105 return raw_origin 

106 

107 try: 

108 parsed = urlsplit(raw_origin) 

109 scheme = parsed.scheme.lower() 

110 # 读取 hostname / port 时才校验 netloc：畸形 IPv6 字面量由 urlsplit() 

111 # 抛出，非数字端口与超出 0-65535 的端口在读取 port 时抛出。 

112 host = parsed.hostname 

113 port = parsed.port 

114 except ValueError: 

115 return None 

116 

117 # 只有 http/https 具备浏览器 Origin 语义，且两者都有可省略的默认端口。 

118 if scheme not in _DEFAULT_PORTS or not host: 

119 return None 

120 if any(character.isspace() for character in host): 120 ↛ 121line 120 didn't jump to line 121 because the condition on line 120 was never true

121 return None 

122 

123 # 显式小写，不依赖 urlsplit 在不同 Python 版本上的大小写处理。 

124 host = host.lower() 

125 # 浏览器序列化 Origin 时给 IPv6 字面量保留方括号；``:443``、``:0443``、 

126 # ``:80`` 这类默认端口写法必须折叠成不带端口的形态才能与真实前端匹配。 

127 if ":" in host: 

128 host = f"[{host}]" 

129 if port is None or port == _DEFAULT_PORTS[scheme]: 

130 return f"{scheme}://{host}" 

131 return f"{scheme}://{host}:{port}"

134def parse_cors_allowed_origins(raw_origins: str | None) -> list[str]: 

135 """解析浏览器跨域来源白名单。 

136 

137 CORS 只约束浏览器中的跨域 JavaScript，不影响 curl、Postman、n8n 

138 或服务端 SDK。未配置时返回空列表，表示默认不开放跨域访问；用户确实 

139 部署了独立网页前端时，再通过 ``CORS_ALLOWED_ORIGINS`` 显式开启。 

140 

141 每个来源都先折叠成 Origin 请求头的规范形式，因此 ``https://a.example/``、 

142 ``HTTPS://A.Example`` 与显式默认端口的 ``https://a.example:443`` 得到同一个 

143 结果，无法匹配任何 Origin 的写法会被丢弃并留下告警。畸形条目只丢弃自身， 

144 不会中断这里的解析——本函数在模块导入期调用，解析异常会让 API 无法启动。 

145 """ 

146 

147 if not raw_origins: 

148 return [] 

149 

150 # 去除逗号分隔项两侧的空白，并忽略空项，避免常见的环境变量格式 

151 # ``https://a.example, https://b.example,`` 产生永远无法匹配的来源。 

152 origins: list[str] = [] 

153 for candidate in raw_origins.split(","): 

154 item = candidate.strip() 

155 if not item: 

156 continue 

157 origin = _normalize_allowed_origin(item) 

158 if origin is None: 

159 logger.warning( 

160 f"ignoring configured CORS origin that cannot match a browser " 

161 f"Origin header: {item!r}" 

162 ) 

163 continue 

164 if origin not in origins: 

165 origins.append(origin) 

166 return origins

205def is_browser_origin_allowed( 

206 request: Request, allowed_origins: list[str] 

207) -> bool: 

208 """判断浏览器请求来源是否为同源或显式白名单来源。""" 

209 

210 origin = request.headers.get("origin") 

211 if not origin: 

212 # curl、Postman、n8n 和服务端 SDK 通常不发送 Origin。保留这类请求， 

213 # 避免安全修复错误地改变现有 API 客户端的调用契约。 

214 return True 

215 if "*" in allowed_origins or origin in allowed_origins: 

216 return True 

217 

218 # 浏览器对同源 POST 也可能发送 Origin。仅比较 scheme + authority，忽略 

219 # 路径和查询参数；反向代理部署若未正确转发公网 scheme/host，可通过显式 

220 # CORS_ALLOWED_ORIGINS 声明外部来源，避免依赖不可信的转发 Header。 

221 request_url = urlsplit(str(request.url)) 

222 request_origin = f"{request_url.scheme}://{request_url.netloc}" 

223 return origin == request_origin
