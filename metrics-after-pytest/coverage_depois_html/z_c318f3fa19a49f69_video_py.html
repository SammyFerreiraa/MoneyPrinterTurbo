142def _scale_subtitle_frame_on_canvas(frame: np.ndarray, scale: float) -> np.ndarray:

147 相同的缩放与裁剪，否则动画首帧会把透明区域当成黑色文字轮廓合成到视频 

148 上。二维数组表示取值为 0～1 的蒙版，三维数组表示 RGB/RGBA 颜色帧。 

149 """

151 raise ValueError("subtitle frame must be a 2D mask or 3D color frame")

153 height, width = frame.shape[:2] 

154 scaled_width = max(1, int(round(width * scale))) 

155 scaled_height = max(1, int(round(height * scale))) 

156 offset = ((width - scaled_width) // 2, (height - scaled_height) // 2) 

157 

158 if frame.ndim == 2: 

159 # MoviePy 蒙版使用 0～1 浮点数，Pillow 的 L 模式使用 0～255；转换后

165 (scaled_width, scaled_height), 

166 Image.Resampling.BILINEAR, 

167 )

169 mask_canvas.paste(resized_mask, offset)

229 subclipped_items: List[SubClippedVideoClip],

232 source_groups: dict[str, str] | None = None, 

233) -> List[SubClippedVideoClip]: 

234 """ 

235 优先让每个源素材只出现一次，降低成片里同一素材反复出现的概率。 

236 

237 线上素材经常会遇到“一个长视频被切成多个短片段”的情况。旧逻辑在 

238 random 模式下直接打乱所有短片段，导致同一个源视频的多个切片可能 

239 分布在开头和中间，用户会感知为素材重复。本函数只调整片段顺序： 

240 先放每个源文件里最长的一个片段，剩余片段作为兜底；当素材总时长不足时， 

241 仍然允许后续片段补齐音频长度，避免破坏视频生成成功率。优先选择最长 

242 片段是为了避免随机选中视频尾部的零碎短片段，导致明明有足够素材却过早复用。 

243 """

245 return [] 

246 

247 concat_mode_value = getattr(concat_mode, "value", concat_mode) 

248 if concat_mode_value != VideoConcatMode.random.value: 

249 if source_usage is None: 

250 return subclipped_items

252 return sorted( 

253 subclipped_items, 

254 key=lambda item: source_usage.get(item.source_file_path, 0), 

255 ) 

256 # Keep keyword rounds in order while rotating candidates within each keyword. 

257 groups = {} 

258 for item in subclipped_items: 

259 key = source_groups.get(item.source_file_path, item.source_file_path) 

260 groups.setdefault(key, []).append(item) 

261 for items in groups.values(): 

262 items.sort(key=lambda item: source_usage.get(item.source_file_path, 0)) 

263 return [ 

264 item 

265 for row in itertools.zip_longest(*groups.values()) 

266 for item in row 

267 if item is not None 

268 ] 

269

271 for item in subclipped_items: 

272 grouped_items.setdefault(item.source_file_path, []).append(item) 

273 

274 primary_items = [] 

275 overflow_items = [] 

276 for items in grouped_items.values(): 

277 primary_item = max(items, key=lambda item: item.duration) 

278 primary_items.append(primary_item)
